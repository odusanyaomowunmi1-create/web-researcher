<body>
  <div class="container">
    <h1>Research anything on the web</h1>
    <p class="description">Give me a question or topic and I'll find the most relevant information.</p>

    <div class="input-group">
      <textarea id="query" placeholder="What do you want me to research?"></textarea>
      <button id="researchBtn">Research</button>
    </div>

    <div id="status"></div>
    <div id="results"></div>
  </div>

  <script>
    const btn = document.getElementById('researchBtn');
    const queryInput = document.getElementById('query');
    const statusEl = document.getElementById('status');
    const resultsEl = document.getElementById('results');

    btn.addEventListener('click', async () => {
      const query = queryInput.value.trim();
      if (!query) {
        showStatus('Please enter a research question.', true);
        return;
      }

      btn.disabled = true;
      btn.textContent = 'Researching...';
      resultsEl.innerHTML = '';
      showStatus('Researching the web...', false);

      try {
        const res = await fetch('/.netlify/functions/research', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ query })
        });

        const data = await res.json();

        if (!res.ok) {
          throw new Error(data.error || 'Request failed');
        }

        statusEl.classList.remove('visible');
        renderResults(data.results || []);
      } catch (err) {
        showStatus(err.message || 'Something went wrong. Please try again.', true);
      } finally {
        btn.disabled = false;
        btn.textContent = 'Research';
      }
    });

    function showStatus(message, isError) {
      statusEl.textContent = message;
      statusEl.classList.add('visible');
      statusEl.classList.toggle('error', isError);
    }

    function renderResults(results) {
      if (!results.length) {
        showStatus('No results found.', true);
        return;
      }

      resultsEl.innerHTML = results.map(r => `
        <div class="result-card">
          <div class="result-title">${escapeHtml(r.title || 'Untitled')}</div>
          <div class="result-source">Source: ${escapeHtml(r.site_name || 'Unknown')}</div>
          <a class="result-url" href="${escapeHtml(r.url)}" target="_blank" rel="noopener">
            ${escapeHtml(r.url)}
          </a>
          <p class="result-description">${escapeHtml(r.snippet || '')}</p>
        </div>
      `).join('');
    }

    function escapeHtml(text) {
      const div = document.createElement('div');
      div.textContent = text;
      return div.innerHTML;
    }

    queryInput.addEventListener('keydown', (e) => {
      if (e.key === 'Enter' && (e.ctrlKey || e.metaKey)) {
        btn.click();
      }
    });
  </script>
